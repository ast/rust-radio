* Sideband

The super simple Opus RTP stream.

* Building

#+begin_src bash
  # Install cross
  cargo install cross --git https://github.com/cross-rs/cross
  cross --version --verbose

  # You have to docker login to ghcr.io or it will fail
  # For a 64-bit Pi 4/5
  cross build --target aarch64-unknown-linux-gnu --release

  # There is a Justfile in the workspace
  cargo install just

  # Then you can do
  just build
  just upload
#+end_src

* Testing

#+begin_src bash
  sudo apt update
  sudo apt install \
      libgstreamer1.0-dev \
      libgstreamer-plugins-base1.0-dev \
      libgstreamer-plugins-bad1.0-dev \
      gstreamer1.0-plugins-base \
      gstreamer1.0-plugins-good \

      gstreamer1.0-libav \
      gstreamer1.0-tools \

  # Inspect installtion
  gst-inspect-1.0 opusdec
  gst-inspect-1.0 rtpopusdepay
#+end_src

You can receive the audio stream using =gstreamer=.

#+begin_src bash
  # Level 2 = Errors and Warnings only
  export GST_DEBUG=2
  # Level 5 = LOG (Very detailed)
  export GST_DEBUG=5

  gst-launch-1.0 -v \
                   udpsrc port=5004 caps="application/x-rtp, media=(string)audio, clock-rate=(int)48000, encoding-name=(string)OPUS, payload=(int)96" ! \
                   rtpopusdepay ! opusdec ! audioconvert ! level ! fakesink
#+end_src
